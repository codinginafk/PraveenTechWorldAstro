## 2. The Full Python Content Auditor Script

**Run this standalone Python script against any Markdown or MDX file to generate an instantaneous 7-dimension score and defect report.**

Save this file as `scripts/content_auditor.py`:

```python
#!/usr/bin/env python3
"""
scripts/content_auditor.py
PraveenTechWorld Quality Scoring Engine
Evaluates Markdown/MDX content across 7 statistical linguistic dimensions.
"""

import sys
import re
import math
from collections import Counter

AI_BUZZWORDS = {
    "delve", "tapestry", "seamless", "revolutionize", "testament",
    "foster", "intertwined", "moreover", "furthermore", "pivotal",
    "beacon", "harness", "multifaceted", "underpin", "paramount"
}

def clean_markdown(text: str) -> str:
    """Strip frontmatter, code blocks, and markdown syntax."""
    text = re.sub(r'^---[\s\S]*?---', '', text)          # Frontmatter
    text = re.sub(r'```[\s\S]*?```', '', text)            # Fenced code
    text = re.sub(r'`[^`]*`', '', text)                   # Inline code
    text = re.sub(r'\[([^\]]+)\]\([^\)]+\)', r'\1', text) # Links
    text = re.sub(r'[#*_>~]', '', text)                   # Formatting characters
    return text.strip()

def analyze_content(file_path: str):
    with open(file_path, 'r', encoding='utf-8') as f:
        raw_text = f.read()

    cleaned = clean_markdown(raw_text)
    words = re.findall(r'\b[a-zA-Z]{2,}\b', cleaned.lower())
    sentences = [s.strip() for s in re.split(r'[.!?]+', cleaned) if len(s.strip()) > 3]

    if not words or not sentences:
        print("[!] File does not contain enough text for analysis.")
        return

    word_count = len(words)
    sentence_count = len(sentences)

    # 1. Lexical Diversity (Type-Token Ratio)
    unique_words = len(set(words))
    ttr = unique_words / word_count

    # 2. Sentence Length Standard Deviation (Burstiness)
    sentence_lengths = [len(re.findall(r'\b\w+\b', s)) for s in sentences]
    mean_length = sum(sentence_lengths) / sentence_count
    variance = sum((l - mean_length) ** 2 for l in sentence_lengths) / sentence_count
    burstiness = math.sqrt(variance)

    # 3. AI Buzzword Density
    found_buzzwords = [w for w in words if w in AI_BUZZWORDS]
    buzzword_density = (len(found_buzzwords) / word_count) * 100

    # 4. Em-Dash Frequency
    em_dashes = len(re.findall(r'—|--', raw_text))
    em_dash_rate = (em_dashes / word_count) * 1000

    # Composite Score Calculation (0 - 100)
    score_ttr = min(100, max(0, (ttr - 0.20) / 0.25 * 100))
    score_burst = min(100, max(0, (burstiness - 4.0) / 8.0 * 100))
    score_buzz = max(0, 100 - (buzzword_density * 200))
    score_dash = max(0, 100 - max(0, (em_dash_rate - 1.0) * 20))

    composite_score = round(
        (score_ttr * 0.30) +
        (score_burst * 0.30) +
        (score_buzz * 0.20) +
        (score_dash * 0.20), 1
    )

    print("\n" + "="*55)
    print(f" CONTENT QUALITY AUDIT REPORT: {file_path}")
    print("="*55)
    print(f" Word Count:                   {word_count}")
    print(f" Sentence Count:               {sentence_count}")
    print(f" Lexical Diversity (TTR):      {ttr:.3f} (Target: >= 0.380)")
    print(f" Sentence Burstiness (StdDev): {burstiness:.2f} words (Target: >= 8.50)")
    print(f" Flagged AI Buzzwords:         {len(found_buzzwords)} ({buzzword_density:.2f}%)")
    print(f" Em-Dash Count:                {em_dashes} ({em_dash_rate:.1f} / 1k words)")
    print("-" * 55)
    print(f" COMPOSITE QUALITY SCORE:      {composite_score} / 100")
    print("="*55 + "\n")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python content_auditor.py <path-to-article.mdx>")
        sys.exit(1)
    analyze_content(sys.argv[1])
```

## 4. Automated Cleanup: The Em-Dash & Header Regex Script

**Manually editing 253 em dashes across 69 files takes hours. Our team automated the remediation using a safe AST regex script that protects code blocks and frontmatter.**

Save this script as `scripts/clean_emdash.py`:

```python
#!/usr/bin/env python3
"""
scripts/clean_emdash.py
Replaces ornamental em dashes with structured punctuation while protecting code blocks.
"""

import sys
import re

def clean_em_dashes(file_path: str):
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()

    # Split frontmatter, body, and code blocks
    parts = re.split(r'(```[\s\S]*?```)', content)
    modified = False
    cleaned_parts = []

    for part in parts:
        if part.startswith('```'):
            cleaned_parts.append(part) # Preserve code block verbatim
        else:
            # Rule 1: Replace bold lead dashes with colons (**Feature** — text -> **Feature:** text)
            new_part = re.sub(r'(\*\*[^*]+\*\*)\s*—\s*', r'\1: ', part)
            # Rule 2: Replace orphaned space dashes with clean semicolons or commas
            new_part = re.sub(r'\s+—\s+', ', ', new_part)
            if new_part != part:
                modified = True
            cleaned_parts.append(new_part)

    if modified:
        with open(file_path, 'w', encoding='utf-8') as f:
            f.write("".join(cleaned_parts))
        print(f"[FIXED ✅] Sanitized em dashes in: {file_path}")
    else:
        print(f"[OK ⚪] No em dash violations in: {file_path}")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python clean_emdash.py <file.mdx>")
        sys.exit(1)
    clean_em_dashes(sys.argv[1])
```

## Decision Summary: How to Audit Your Own Blog

- **Run statistical audits before publishing:** Never rely solely on an LLM to evaluate text quality. Use `content_auditor.py` to establish objective mathematical baselines.
- **Enforce lexical diversity $\ge 0.38$:** If your Type-Token Ratio dips below 0.35, actively rephrase repetitive verbs and expand descriptive technical vocabulary.
- **Vary sentence lengths intentionally:** Mix 5-word declarative sentences with 25-word compound explanations to create natural reading cadence.
