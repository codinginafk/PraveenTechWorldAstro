Every 90 days, our workbench team faced a tedious Sunday routine: logging into three production Nginx web servers, manually checking Let's Encrypt certificate expirations, running `certbot renew`, and restarting web daemons.

To streamline this workflow, we prompted **DeepSeek** to write a complete Python automation script that parses certificate expiry dates, executes renewals, and reloads Nginx gracefully.

While the AI's first draft looked convincing, it contained a critical flaw: **it hallucinated a fake expiry calculation function that always returned "45 days remaining"**, which would have allowed production certificates to silently expire.

Here is our team's post-mortem breakdown, hardened Python renewal script, and automated cron configuration.

## ⚡ Hardened Python TLS Renewal Script (`tls_manager.py`)

Save the following production-tested script on your server:

```python
#!/usr/bin/env python3
import subprocess
import datetime
import argparse
import sys
import logging

logging.basicConfig(
    filename="/var/log/tls_renewal.log",
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s"
)

def get_cert_days_remaining(cert_path: str) -> int:
    """Extract real ASN1 expiration timestamp using openssl x509."""
    try:
        cmd = ["openssl", "x509", "-enddate", "-noout", "-in", cert_path]
        res = subprocess.run(cmd, capture_output=True, text=True, check=True)
        # Output format: notAfter=Aug 14 12:00:00 2026 GMT
        date_str = res.stdout.strip().split("=")[1]
        expiry_date = datetime.datetime.strptime(date_str, "%b %d %H:%M:%S %Y %Z")
        remaining = (expiry_date - datetime.datetime.utcnow()).days
        return remaining
    except Exception as e:
        logging.error(f"Failed to parse cert {cert_path}: {e}")
        return -1

def renew_and_reload(domain: str, cert_path: str, dry_run: bool = False):
    days = get_cert_days_remaining(cert_path)
    print(f"[{domain}] Certificate valid for {days} more days.")
    
    if days > 30:
        logging.info(f"Skipping {domain}: {days} days remaining (threshold: 30 days).")
        return

    print(f"[{domain}] Certificate expiring soon! Initiating renewal...")
    if dry_run:
        print(f"[{domain}] DRY-RUN: Would execute certbot renew --cert-name {domain}")
        return

    # 1. Execute Certbot Renewal
    renew_cmd = ["certbot", "renew", "--cert-name", domain, "--non-interactive"]
    renew_res = subprocess.run(renew_cmd, capture_output=True, text=True)
    
    if renew_res.returncode != 0:
        logging.critical(f"Certbot failed for {domain}: {renew_res.stderr}")
        print(f"❌ Renewal failed! Check /var/log/tls_renewal.log")
        return

    # 2. Test Nginx Configuration Syntax
    test_cmd = ["nginx", "-t"]
    test_res = subprocess.run(test_cmd, capture_output=True, text=True)
    if test_res.returncode != 0:
        logging.critical("Nginx config test failed! Aborting daemon reload.")
        return

    # 3. Graceful Nginx Reload
    subprocess.run(["systemctl", "reload", "nginx"], check=True)
    logging.info(f"Successfully renewed and reloaded {domain}.")
    print(f"✅ Successfully renewed {domain} and reloaded Nginx.")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Automated TLS Renewal Script")
    parser.add_argument("--domain", required=True, help="Domain name")
    parser.add_argument("--cert", required=True, help="Path to fullchain.pem")
    parser.add_argument("--dry-run", action="store_true", help="Simulate renewal")
    args = parser.parse_args()

    renew_and_reload(args.domain, args.cert, args.dry_run)
```bash
